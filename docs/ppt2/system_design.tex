\section{数据库设计}
% ================================================================

\begin{frame}{设计目标与通用约定}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{block}{设计原则}
            \begin{enumerate}
                \item \textbf{强一致优先}：核心业务由 PostgreSQL 保证 ACID
                \item \textbf{大文件分离}：图片走对象存储，DB 仅存元数据
                \item \textbf{计数器冗余}：避免高频 COUNT 聚合
                \item \textbf{软删除统一}：\texttt{deleted\_at}，数据可追溯
            \end{enumerate}
        \end{block}
        \column{0.5\textwidth}
        \begin{block}{通用约定}
            \small
            \begin{tabularx}{\linewidth}{lX}
                \toprule
                \textbf{约定} & \textbf{说明} \\
                \midrule
                主键 & UUID，防枚举 \\
                时间戳 & TIMESTAMPTZ NOT NULL \\
                软删除 & \texttt{deleted\_at} NULL \\
                计数器 & BIGINT DEFAULT 0 \\
                状态字段 & VARCHAR + CHECK \\
                \bottomrule
            \end{tabularx}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{实体关系概览}
    \centering
    \includegraphics[height=0.9\textheight]{er_diagram.png}
\end{frame}

\begin{frame}{核心表设计要点}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{block}{\texttt{users} — 用户表}
            \begin{itemize}
                \item \texttt{username} VARCHAR(32), UNIQUE
                \item \texttt{email} VARCHAR(255), UNIQUE
                \item \texttt{password\_hash} — pwdlib (Argon2)
                \item \texttt{role} — CHECK IN ('user','admin')
                \item \texttt{status} — CHECK IN ('active','inactive','banned')
                \item \texttt{email\_verified} BOOLEAN
            \end{itemize}
        \end{block}
        \column{0.5\textwidth}
        \begin{block}{\texttt{posts} — 帖子表}
            \begin{itemize}
                \item \texttt{board\_id / author\_id} — FK
                \item \texttt{content} TEXT（支持 Markdown）
                \item \texttt{status} — CHECK IN ('normal','hidden','deleted')
                \item \texttt{is\_pinned / is\_featured}
                \item \texttt{like\_count / comment\_count / view\_count}
            \end{itemize}
        \end{block}
    \end{columns}
    \vspace{0.2cm}
    \begin{block}{\texttt{notifications} — 通知表}
        \begin{itemize}
            \item \texttt{type} CHECK IN ('comment','reply','like','system')
            \item \texttt{is\_read} BOOLEAN + \texttt{read\_at} TIMESTAMPTZ
            \item 触发规则：评论帖子→通知作者、回复评论→通知被回复者、点赞→通知目标作者
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{索引策略与关键决策}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{核心索引}
            \begin{itemize}
                \item \texttt{posts(board\_id, is\_pinned DESC, created\_at DESC)}
                \item \texttt{notifications(recipient\_id, is\_read, created\_at)}
                \item \texttt{comments(post\_id)}, \texttt{comments(parent\_comment\_id)}
                \item \texttt{post\_likes(post\_id, user\_id)} UNIQUE
                \item 所有 \texttt{deleted\_at} 建索引
            \end{itemize}
        \end{block}
        \column{0.52\textwidth}
        \begin{block}{关键设计决策}
            \small
            \begin{tabularx}{\linewidth}{lX}
                \toprule
                \textbf{决策} & \textbf{理由} \\
                \midrule
                UUID 主键 & 防枚举 \\
                软删除 & 数据可追溯可恢复 \\
                计数器冗余 & 换列表免 COUNT 聚合 \\
                VARCHAR+CHECK & 避免 ENUM 迁移成本 \\
                评论双外键 & 一次查询获取所有楼层 \\
                通知异步化 & 不阻塞用户请求 \\
                \bottomrule
            \end{tabularx}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{KP-1 \& KP-2：认证与鉴权链路}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{KP-1 登录流程}
            \begin{enumerate}
                \item 客户端提交账号与密码
                \item 按用户名或邮箱查询用户
                \item \texttt{verify\_password()} 校验
                \item 检查 \texttt{status == active}
                \item 更新 \texttt{last\_login\_at}
                \item 生成 JWT（含 sub, role, exp）
                \item 返回 \texttt{ApiResponse<LoginData>}
            \end{enumerate}
        \end{block}
        \column{0.52\textwidth}
        \includegraphics[width=\textwidth]{auth_decision.png}
    \end{columns}
\end{frame}

\begin{frame}{KP-3 \& KP-6：发布帖子与管理封禁}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{block}{KP-3 发布帖子}
            \textbf{流程：}
            \begin{enumerate}
                \item \textbf{请求准入}：\texttt{get\_current\_user()} 鉴权
                \item \textbf{输入校验}：Pydantic \texttt{PostCreate}
                \item \textbf{数据封装}：合并业务字段 + author\_id
                \item \textbf{持久化}：add $\rightarrow$ commit $\rightarrow$ refresh
                \item \textbf{响应输出}：\texttt{ApiResponse<PostRead>}
            \end{enumerate}
            \textbf{异常}：未登录 401、被封禁 403、校验失败 422
        \end{block}
        \column{0.5\textwidth}
        \begin{block}{KP-6 管理员封禁用户}
            \textbf{流程：}
            \begin{enumerate}
                \item \textbf{权限准入}：\texttt{require\_admin()} 双重校验
                \item \textbf{输入校验}：UUID 格式 + 状态正则校验
                \item \textbf{实体定位}：按 uuid 查询用户
                \item \textbf{状态变更}：\texttt{status := "banned"}
                \item \textbf{响应输出}：\texttt{AdminUserData}
            \end{enumerate}
            \textbf{效果}：登录被拒 + 现有 Token 下次请求拦截
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{KP-4：评论/回复与通知联动}
    \begin{block}{评论/回复流程}
        \begin{enumerate}
            \item 校验目标帖子存在且未被删除
            \item 若 \texttt{parent\_comment\_id} 为空 $\rightarrow$ 创建一级评论，\texttt{root\_comment\_id} 指向自身
            \item 若不为空 $\rightarrow$ 继承根评论编号，创建回复
            \item 同步更新 \texttt{post.comment\_count} / \texttt{parent.reply\_count}
            \item 调用 \texttt{NotificationService} 创建通知
        \end{enumerate}
    \end{block}
    \begin{exampleblock}{楼中楼树机制}
        一级评论：\texttt{parent\_comment\_id IS NULL}，\texttt{root\_comment\_id} 指向自身；子回复：\texttt{parent\_comment\_id} 指向被回复评论，\texttt{root\_comment\_id} 指向根楼层
    \end{exampleblock}
\end{frame}

\begin{frame}{KP-5：点赞/取消点赞与计数同步}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{block}{点赞流程}
            \begin{enumerate}
                \item 确认目标帖子/评论存在
                \item 检查未点过（UNIQUE 约束防重）
                \item 新增 Like 记录
                \item \texttt{like\_count + 1}
                \item 创建 like 通知
            \end{enumerate}
        \end{block}
        \column{0.5\textwidth}
        \begin{block}{取消点赞流程}
            \begin{enumerate}
                \item 定位现有点赞记录
                \item 删除记录
                \item \texttt{like\_count - 1}（不低于 0）
            \end{enumerate}
        \end{block}
    \end{columns}
    \vspace{0.2cm}
    \begin{alertblock}{关键约束}
        在同一事务内更新记录+计数（保证一致性）\textbullet{} UNIQUE 约束保证幂等 \textbullet{} 计数不得小于零
    \end{alertblock}
\end{frame}

\section{可靠性及安全性设计}
% ================================================================

\begin{frame}{可靠性设计：数据一致性与服务可用性}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{数据一致性保证}
            \begin{enumerate}
                \item \textbf{事务原子性}：点赞=写入记录+更新计数同时成功/回滚
                \item \textbf{幂等约束}：\texttt{UNIQUE(post\_id, user\_id)} 防重
                \item \textbf{原子更新计数}：数据库表达式更新，约束 $\ge$ 0
                \item \textbf{软删除保护}：不物理删除，支持恢复与审计
                \item \textbf{外键完整性}：帖子必须关联存在的板块和作者
            \end{enumerate}
        \end{block}
        \column{0.52\textwidth}
        \begin{block}{服务可用性策略}
            \begin{enumerate}
                \item \textbf{健康检查}：pg\_isready + redis-cli ping
                \item \textbf{Redis 故障降级}：缓存查询回退 PostgreSQL
                \item \textbf{连接池控制}：设置大小、超时、回收时间
                \item \textbf{超时与重试}：查询允许短重试，写入不盲目重试
                \item \textbf{缓存策略}：更新 DB 后删缓存、空值防穿透
            \end{enumerate}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{安全性设计：身份认证与权限控制}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{身份认证}
            \begin{itemize}
                \item \textbf{密码存储}：pwdlib (Argon2) 加盐哈希
                \item \textbf{JWT 令牌}：含 sub/role/exp，访问受保护接口必须携带
                \item \textbf{会话管理}：Redis 黑名单，登出/封禁即时吊销
                \item \textbf{登录保护}：连续失败限制（429），防暴力破解
            \end{itemize}
        \end{block}
        \column{0.52\textwidth}
        \begin{block}{三层权限控制}
            \small
            \begin{tabularx}{\linewidth}{lX}
                \toprule
                \textbf{层级} & \textbf{机制} \\
                \midrule
                接口级鉴权 & 校验 JWT；管理接口校验 admin \\
                资源级鉴权 & 用户仅操作自己的内容 \\
                状态级鉴权 & banned 禁止一切操作 \\
                \bottomrule
            \end{tabularx}
            \vspace{0.2cm}
            \small
            \begin{tabularx}{\linewidth}{lX}
                \toprule
                \textbf{角色} & \textbf{权限范围} \\
                \midrule
                访客 & 浏览公开帖子 \\
                普通用户 & 发帖/评论/点赞/举报 \\
                管理员 & 用户/板块/公告管理 + 审核 \\
                \bottomrule
            \end{tabularx}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{安全性设计：输入防护与文件上传}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{输入校验与注入防护}
            \begin{itemize}
                \item \textbf{Pydantic 校验}：所有入参类型/长度/格式校验
                \item \textbf{SQL 注入防护}：SQLAlchemy ORM 参数化查询
                \item \textbf{XSS 防护}：服务端白名单过滤 HTML
                \item \textbf{内容长度限制}：标题/评论等字段上限
                \item \textbf{错误信息脱敏}：不暴露 SQL、路径、堆栈
            \end{itemize}
        \end{block}
        \column{0.52\textwidth}
        \begin{block}{文件上传安全}
            \begin{itemize}
                \item \textbf{类型白名单}：png / jpeg / webp / pdf
                \item \textbf{大小限制}：头像/帖子图片/附件分别设置上限
                \item \textbf{文件名安全}：存储 Key 使用 UUID/哈希
                \item \textbf{哈希去重}：sha256 校验，避免重复存储
                \item \textbf{访问控制}：私有文件生成临时访问链接
            \end{itemize}
        \end{block}
    \end{columns}
    \vspace{0.1cm}
    \begin{alertblock}{管理后台安全要点}
        管理员操作二次校验 \textbullet{} \texttt{admin\_audit\_logs} 全量记录 \textbullet{} 举报处理留痕 \textbullet{} 封禁即时生效 \textbullet{} 后台接口隔离（\texttt{/api/v1/admin}）
    \end{alertblock}
\end{frame}
